        // --- 1. ЛІЧИЛЬНИК ДНІВ (Рівно 1095) ---
        const diffDays = 1095; 
        
        let currentDay = 0;
        const counterElement = document.getElementById("days-counter");
        const timer = setInterval(() => {
            if (currentDay >= diffDays) {
                clearInterval(timer);
                counterElement.innerText = diffDays;
            } else {
                currentDay += 15;
                if(currentDay > diffDays) currentDay = diffDays;
                counterElement.innerText = currentDay;
            }
        }, 30);

        // --- 2. ПАДАЮЧІ СЕРДЕЧКА ---
        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart');
            const hearts = ['❤️', '💖', '💕', '✨'];
            heart.innerHTML = hearts[Math.floor(Math.random() * hearts.length)];
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = Math.random() * 3 + 4 + 's';
            heart.style.fontSize = Math.random() * 1 + 1 + 'rem';
            document.body.appendChild(heart);
            setTimeout(() => heart.remove(), 7000);
        }
        setInterval(createHeart, 600);

        let availableReasons = [...allReasons];
        const totalReasonsCount = allReasons.length;

        function updateCounterText() {
            const counterText = document.getElementById('reasons-left-text');
            const shownCount = totalReasonsCount - availableReasons.length;
            if (availableReasons.length === 0) {
                counterText.innerText = `Ти переглянула всі ${totalReasonsCount} причин! Починаємо спочатку ❤️`;
            } else {
                counterText.innerText = `Відкрито причин: ${shownCount} з ${totalReasonsCount}`;
            }
        }

        updateCounterText();

        function generateReason() {
            const display = document.getElementById('reason-display');
            display.style.opacity = 0;
            
            setTimeout(() => {
                if (availableReasons.length === 0) {
                    availableReasons = [...allReasons];
                }

                const randomIndex = Math.floor(Math.random() * availableReasons.length);
                const newReason = availableReasons[randomIndex];

                availableReasons.splice(randomIndex, 1);

                display.innerText = newReason;
                display.style.transition = "opacity 0.5s";
                display.style.opacity = 1;
                
                updateCounterText();
            }, 300);
        }
